/// start doc usr command
void cmd_doc(TomlChIt *it) {
    // initialize struct where doc tags will be accumulated
    DocItemList *dil;
    DocItemList_init(&dil);

    // debug print to check if I've gone insane or not
    // DocItemList_print(dil);

    char *doc_dir_dest = NULL;
    char *doc_dir_target = NULL;
    char *doc_open = NULL;
    if (TomlChIt_get_string(it, "toolkit", "doc_dir_target", &doc_dir_target) !=
            TOML_OK) {
        ERROR("key \"doc_dir_target\" not found in cenv.toml");
    }
    if (TomlChIt_get_string(it, "toolkit", "doc_dir_dest", &doc_dir_dest) !=
            TOML_OK) {
        ERROR("key \"doc_dir_dest\" not found in cenv.toml");
    }
    if (TomlChIt_get_string(it, "toolkit", "doc_open", &doc_open) != TOML_OK) {
        ERROR("key \"doc_open\" not found in cenv.toml");
    }

    // parse and generate
    parse(doc_dir_target, dil);
    gen_html(doc_dir_dest, dil);
    // free
    DocItemList_free(dil);

    if (contains(doc_open, "y")) {
        char cmd[CMD_BUF * 4];
        cmd[0] = '\0';
#ifdef __APPLE__
        strncat(cmd, "open ", CMD_BUF);

#elif defined(__linux__)
        strncat(cmd, "xdg-open ", size);

#else
        ERROR("Unknown or non-unix platform detected.\n");
#endif
        char path[CMD_BUF * 2];
        snprintf(path, sizeof(path), "%s/index.html", doc_dir_dest);
        strncat(cmd, path, CMD_BUF);
        system(cmd);
    }
}

/// start todo usr command
void cmd_todo(TomlChIt *it) {
    const char **todo_dirs = NULL;
    size_t todo_dirs_count = 0;
    if (TomlChIt_get_string_array(it, "toolkit", "todo_dirs", &todo_dirs,
                &todo_dirs_count) != TOML_OK) {
        ERROR("key \"todo_dirs\" not found in cenv.toml");
    }

    for (size_t i = 0; i < todo_dirs_count; i++) {
        char *target = strdup(todo_dirs[i]);
        search(target);
        free(target);
    }
}
