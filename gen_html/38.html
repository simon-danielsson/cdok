void cmd_debug(TomlChIt *it) {
    // ------------------------------------------------
    // git info

    char git_head_hash_short[CMD_BUF];
    char git_head_hash_long[CMD_BUF];
    int status = system("git rev-parse --is-inside-work-tree > /dev/null 2>&1");
    if (status == 0) {
        run_cmd(git_head_hash_short, "git rev-parse --short HEAD");
        run_cmd(git_head_hash_long, "git rev-parse HEAD");
    } else {
        strcpy(git_head_hash_short, "nogit");
        strcpy(git_head_hash_long, "0.0.0");
    }

    // ------------------------------------------------
    // extra flags for environment variables

    char *env_name = NULL;
    if (TomlChIt_get_string(it, "project", "name", &env_name) != TOML_OK) {
        ERROR("key \"name\" not found in cenv.toml");
    }
    char *env_version = NULL;
    if (TomlChIt_get_string(it, "project", "version", &env_version) != TOML_OK) {
        ERROR("key \"version\" not found in cenv.toml");
    }
    char *env_repo = NULL;
    if (TomlChIt_get_string(it, "project", "repo", &env_repo) != TOML_OK) {
        ERROR("key \"repo\" not found in cenv.toml");
    }
    char *env_author = NULL;
    if (TomlChIt_get_string(it, "project", "author", &env_author) != TOML_OK) {
        ERROR("key \"author\" not found in cenv.toml");
    }
    char *env_contact = NULL;
    if (TomlChIt_get_string(it, "project", "contact", &env_contact) != TOML_OK) {
        ERROR("key \"contact\" not found in cenv.toml");
    }
    char *env_c_standard = NULL;
    if (TomlChIt_get_string(it, "project", "c_standard", &env_c_standard) !=
            TOML_OK) {
        ERROR("key \"c_standard\" not found in cenv.toml");
    }

    // ------------------------------------------------
    // make sure that the build dir is there

    char *build_dir = NULL;
    if (TomlChIt_get_string(it, "debug", "build_dir", &build_dir) != TOML_OK) {
        ERROR("key \"build_dir\" not found in cenv.toml");
    }
    char build_dir_temp[CMD_BUF];
    snprintf(build_dir_temp, sizeof(build_dir_temp), "mkdir -p %s", build_dir);
    system(build_dir_temp);

    // ------------------------------------------------
    // bootstrap nob

    char *nob_src_dir = NULL;
    if (TomlChIt_get_string(it, "toolkit", "nob_src_dir", &nob_src_dir) !=
            TOML_OK) {
        ERROR("key \"nob_src_dir\" not found in cenv.toml");
    }

    // cc -o ./tools/nob/nob ./tools/nob/nob.c
    char bootstrap_cmd[CMD_BUF * 2];
    snprintf(bootstrap_cmd, sizeof(bootstrap_cmd), "cc -o %s/nob %s/nob.c",
            nob_src_dir, nob_src_dir);
    system(bootstrap_cmd);

    // ------------------------------------------------
    // build program

    const char **arr = NULL;
    size_t arr_count = 0;
    if (TomlChIt_get_string_array(it, "debug", "c_flags", &arr, &arr_count) !=
            TOML_OK) {
        ERROR("todo_dirs not found in toml");
    }

    // TODO: instead of a command like "release" and "debug" and so on that path
    // should instead just be the path to the src dir (in the case of the debug
    // build it should be the the value of the src field in the [debug] table in
    // cenv.toml)
    // the new build command should look more like this:
    // APP_VERSION="$VERSION" ./tools/nob/nob $src_dir $env_vars $c_flags

    // APP_VERSION="$VERSION" ./tools/nob/nob release ${nob_extra_flags[*]}
    // mv ./build/release/main
    // ./build/release/$project_name-$VERSION-$git_head_hash_short
    // ./build/release/$project_name-$VERSION-$git_head_hash_short
}
