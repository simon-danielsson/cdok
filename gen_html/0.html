/// returns true if chars are the same
/// @param char char
bool cmp_c(char x, char y) { return x == y; }

/// helper to append char
/// @important needed because of null terminator issues
static bool append_char(char *buf, size_t cap, char c) {
    size_t len = strlen(buf);

    if (len + 1 >= cap) {
        return false;
    }

    buf[len] = c;
    buf[len + 1] = '\0';
    return true;
}

/// tokens definition
enum TokenType {
    COMMA,    // ,
    CU_BR_OP, // {
    CU_BR_CL, // };
    SQ_BR_OP, // [
    SQ_BR_CL, // ]
    QUOTE,    // "
    EQUALS,   // =
    SPACE,    // ' '
    NEWLINE,  // \n
    HASH,     // #
};

const char token[] = {
    ',', '{', '}', '[', ']', '"', '=', ' ', '\n', '#',
};

/// push table to toml iterator struct
void TomlChIt_push_table(TomlChIt *tci, Table *t) {
    Table *tmp = realloc(tci->t, (tci->t_count + 1) * sizeof(Table));

    if (!tmp) {
        ERROR("Allocation failure");
        return;
    }

    tci->t = tmp;
    tci->t[tci->t_count] = *t;
    tci->t_count++;
}

/// initalizer for TomlChIt iterator
/// @param TomlChIt struct
/// @param path of input file
bool TomlChIt_init(TomlChIt *it, const char *path) {
    it->file = fopen(path, "r");
    if (!it->file) {
        ERROR("Unable to open file: %s", path);
        return false;
    }
    it->inside_val = false;
    it->inside_header = false;
    it->table_in_prog = false;
    it->inside_key = false;
    it->inside_val_arr = false;
    it->inside_val_arr_str = false;
    it->inside_comment = false;
    it->key_wants_val = false;
    it->t_count = 0;
    it->t = NULL;
    return true;
}

// After a newline (or start of file):
// Skip whitespace
// If next char is:
// # → comment
// [ → table
// ignore whitespace

static void trim_right(char *s) {
    size_t len = strlen(s);

    while (len > 0 &&
            (s[len - 1] == ' ' || s[len - 1] == '\t' || s[len - 1] == '=' ||
             s[len - 1] == '\r' || s[len - 1] == '\n')) {
        s[len - 1] = '\0';
        len--;
    }
}

#define TEMP_BUFF_SIZE 256

/// toml parser
/// uses an iterator to recursively collect a character stream into structures
/// @param path of input file
void parse_toml(TomlChIt *it) {

    // temp values
    char table_header[TEMP_BUFF_SIZE];
    table_header[0] = '\0';
    char key[TEMP_BUFF_SIZE];
    key[0] = '\0';
    char val[TEMP_BUFF_SIZE];
    val[0] = '\0';
    char **val_arr_items = NULL;
    size_t val_arr_items_count = 0;

    // main parsing loop
    while ((it->c = fgetc(it->file)) != EOF) {

        // ---------------------
        // newlines and spaces

        if (cmp_c(it->c, token[NEWLINE])) {
            it->inside_comment = false;
            it->inside_key = false;
            it->key_wants_val = false;
            memset(key, 0, TEMP_BUFF_SIZE);
            // continue;
        }
        if (it->inside_comment) {
            continue;
        }
        if (!it->inside_val && !it->inside_val_arr && !it->inside_val_arr_str &&
                cmp_c(it->c, token[HASH])) {
            it->inside_comment = true;
            continue;
        }
        // if (cmp_c(it->c, token[NEWLINE])) {
        //     it->inside_comment = false;
        // }
        if (cmp_c(it->c, token[SPACE]) && !it->inside_val && !it->inside_val_arr &&
                !it->inside_val_arr_str) {
            continue;
        }

        // ---------------------
        // table header

        if ((!it->key_wants_val) && cmp_c(it->c, token[SQ_BR_OP])) {
            it->inside_header = true;
            if (!it->table_in_prog) {
                it->table_in_prog = true;
            }
            continue;
        }
        if (it->inside_header && cmp_c(it->c, token[SQ_BR_CL])) {
            it->inside_header = false;
            // printf("----- %s -----\n", table_header);
            Table *tmp = realloc(it->t, (it->t_count + 1) * sizeof *it->t);
            if (!tmp) {
                ERROR("Allocation failure");
            }
            it->t = tmp;
            it->t[it->t_count].name = xstrdup(table_header);
            it->t[it->t_count].kvp = NULL;
            it->t[it->t_count].kvp_count = 0;
            it->t_count++;
            memset(table_header, 0, TEMP_BUFF_SIZE);
            continue;
        }
        if (it->inside_header) {
            if (!append_char(table_header, TEMP_BUFF_SIZE, it->c)) {
                ERROR("Header too long");
            }
            continue;
        }

        // ---------------------
        // array values

        if (it->key_wants_val && cmp_c(it->c, token[SQ_BR_OP])) {
            it->inside_val_arr = true;
            continue;
        }

        if (it->inside_val_arr && !it->inside_val_arr_str &&
                cmp_c(it->c, token[SQ_BR_CL])) {

            it->inside_val_arr = false;
            it->key_wants_val = false;

            Table *cur = &it->t[it->t_count - 1];
            KvPair *kv = &cur->kvp[cur->kvp_count - 1];

            kv->value = (Value){.type = VAL_STRING_ARRAY,
                .array = {.items = (const char **)val_arr_items,
                    .count = val_arr_items_count}};

            val_arr_items = NULL;
            val_arr_items_count = 0;
            continue;
        }

        if (it->inside_val_arr_str) {
            if (cmp_c(it->c, token[QUOTE])) {
                it->inside_val_arr_str = false;
                char **tmp = realloc(val_arr_items,
                        (val_arr_items_count + 1) * sizeof *val_arr_items);
                if (!tmp) {
                    ERROR("Allocation failure");
                }
                val_arr_items = tmp;
                val_arr_items[val_arr_items_count++] = xstrdup(val);
                memset(val, 0, sizeof(char) * TEMP_BUFF_SIZE);
                continue;
            } else {
                if (!append_char(val, TEMP_BUFF_SIZE, it->c)) {
                    ERROR("Val too long");
                }
            }
            continue;
        }

        if (it->inside_val_arr && cmp_c(it->c, token[QUOTE])) {
            memset(val, 0, sizeof(char) * TEMP_BUFF_SIZE);
            it->inside_val_arr_str = true;
        }

        // ---------------------
        // values

        if (it->key_wants_val && !it->inside_val && cmp_c(it->c, token[QUOTE])) {
            it->inside_val = true;
            it->key_wants_val = false;
            continue;
        }
        if (it->inside_val && cmp_c(it->c, token[QUOTE])) {
            it->inside_val = false;
            it->key_wants_val = false;
            // printf("%s = %s\n", key, val);
            Table *cur = &it->t[it->t_count - 1];
            KvPair *kv = &cur->kvp[cur->kvp_count - 1];

            kv->value = (Value){.type = VAL_STRING, .str = xstrdup(val)};
            memset(val, 0, sizeof(char) * TEMP_BUFF_SIZE);
            continue;
        }

        if (it->inside_val) {
            if (!append_char(val, TEMP_BUFF_SIZE, it->c)) {
                ERROR("Val too long");
            }
            continue;
        }
        if (it->inside_val_arr && cmp_c(it->c, token[COMMA])) {
            continue;
        }

        // ---------------------
        // keys
        if (!it->inside_val && !it->inside_val_arr) {
            if (it->inside_key && cmp_c(it->c, token[EQUALS])) {
                it->inside_key = false;
                it->key_wants_val = true;

                Table *cur = &it->t[it->t_count - 1];

                KvPair *tmp =
                    realloc(cur->kvp, (cur->kvp_count + 1) * sizeof *cur->kvp);
                if (!tmp) {
                    ERROR("Allocation failure");
                    return;
                }

                cur->kvp = tmp;
                trim_right(key);
                cur->kvp[cur->kvp_count].key = xstrdup(key);
                cur->kvp_count++;

                memset(key, 0, TEMP_BUFF_SIZE);
                continue;
            }
            if (it->inside_key) {
                if (!append_char(key, TEMP_BUFF_SIZE, it->c)) {
                    ERROR("Key too long");
                }

                continue;
            }
            if (!it->key_wants_val) {
                it->inside_key = true;
            }
        }
    }
    fclose(it->file);
}

/// debug print for toml parser iterator
void debug_print(TomlChIt *it) {
    for (size_t i = 0; i < it->t_count; i++) {
        printf("--- %s ---\n", it->t[i].name);

        for (size_t j = 0; j < it->t[i].kvp_count; j++) {
            KvPair *kv = &it->t[i].kvp[j];

            printf("%s = \n", kv->key);

            switch (kv->value.type) {
                case VAL_STRING:
                    printf("%s\n", kv->value.str);
                    break;

                case VAL_STRING_ARRAY:
                    printf("[");
                    for (size_t k = 0; k < kv->value.array.count; k++) {
                        printf("%s%s", kv->value.array.items[k],
                                k + 1 < kv->value.array.count ? ", " : "");
                    }
                    printf("]\n");
                    break;
            }
        }
    }
}
